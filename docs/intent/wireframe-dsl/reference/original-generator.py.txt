def text(x,y,s,size=14,weight=400,color=INK,anchor=None):
    a=f' text-anchor="{anchor}"' if anchor else ''
    return f'<text x="{x}" y="{y}" font-size="{size}" font-weight="{weight}" fill="{color}"{a}>{escape(s)}</text>'
def rect(x,y,w,h,fill=PAPER,stroke=LINE,r=4,dash=False):
    return f'<rect x="{x}" y="{y}" width="{w}" height="{h}" rx="{r}" fill="{fill}" stroke="{stroke}"'+(' stroke-dasharray="6 4"' if dash else '')+'/>'
def line(x1,y1,x2,y2,color=LINE,dash=False):
    return f'<path d="M{x1} {y1}L{x2} {y2}" fill="none" stroke="{color}"'+(' stroke-dasharray="5 4"' if dash else '')+'/>'
def button(x,y,w,label,new=False,active=False,disabled=False):
    return rect(x,y,w,30,SOFT if active else PAPER,BLUE if new or active else LINE,4,new)+text(x+w/2,y+20,label,12,600,'#99a1aa' if disabled else BLUE if active else INK,'middle')
def lines(x,y,items,size=13,step=21,color=INK,weight=400):
    return ''.join(text(x,y+i*step,s,size,weight,color) for i,s in enumerate(items))
def tag(x,y,label,w=73):
    return rect(x,y,w,23,SOFT,BLUE,3)+text(x+w/2,y+16,label,11,600,BLUE,'middle')
def svg(body,title,desc):
    return f'<svg xmlns="http://www.w3.org/2000/svg" width="{W}" height="{H}" viewBox="0 0 {W} {H}" role="img" aria-labelledby="title desc"><title id="title">{escape(title)}</title><desc id="desc">{escape(desc)}</desc><style>text{{font-family:Arial,Helvetica,sans-serif}} .edge{{fill:none;stroke:#8995a1;stroke-width:1.5;marker-end:url(#arrow)}}</style><defs><marker id="arrow" viewBox="0 0 10 10" refX="9" refY="5" markerWidth="6" markerHeight="6" orient="auto-start-reverse"><path d="M0 0L10 5L0 10z" fill="#8995a1"/></marker></defs>'+rect(0,0,W,H,PAPER,LINE,0)+body+'</svg>'
